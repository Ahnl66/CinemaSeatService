cmake_minimum_required(VERSION 3.20)
project(cinemaseat_audio_decoder VERSION 0.1.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CSAD_SANITIZERS "" CACHE STRING "Compiler sanitizer list for host verification")
if(CSAD_SANITIZERS)
  add_compile_options(-fsanitize=${CSAD_SANITIZERS} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${CSAD_SANITIZERS})
endif()
set(CSAD_DEPS "" CACHE PATH "Prefix created by scripts/build_dependencies.py")
if(NOT EXISTS "${CSAD_DEPS}/stamp")
  message(FATAL_ERROR "Build pinned dependencies first; supply -DCSAD_DEPS=<prefix>")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" SERVICE_VERSION)
string(STRIP "${SERVICE_VERSION}" SERVICE_VERSION)
if(NOT SERVICE_VERSION STREQUAL PROJECT_VERSION)
  message(FATAL_ERROR "VERSION and CMake project version differ")
endif()
file(GLOB_RECURSE BUILD_SOURCES CONFIGURE_DEPENDS src/*.cpp src/*.hpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${BUILD_SOURCES} dependencies/manifest.json VERSION)
set(BUILD_HASH_INPUT "")
foreach(SOURCE IN LISTS BUILD_SOURCES)
  file(SHA256 "${SOURCE}" FILE_HASH)
  string(APPEND BUILD_HASH_INPUT "${FILE_HASH}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/manifest.json" DEPENDENCY_HASH)
string(SHA256 CSAD_BUILD_ID "${BUILD_HASH_INPUT}${DEPENDENCY_HASH}")
find_package(Threads REQUIRED)
find_library(AVCODEC avcodec PATHS "${CSAD_DEPS}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(AVUTIL avutil PATHS "${CSAD_DEPS}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(csad_core src/wire.cpp src/config.cpp src/codec_pipeline.cpp)
target_include_directories(csad_core PUBLIC src "${CSAD_DEPS}/include")
target_link_libraries(csad_core PUBLIC ${AVCODEC} ${AVUTIL} Threads::Threads)
target_compile_options(csad_core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(csad_core PUBLIC BOOST_ASIO_NO_DEPRECATED BOOST_ERROR_CODE_HEADER_ONLY)
set(CMAKE_BUILD_RPATH "${CSAD_DEPS}/lib")
set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
add_executable(cinemaseat-audio-decoder src/main.cpp src/server.cpp)
target_link_libraries(cinemaseat-audio-decoder PRIVATE csad_core)
target_compile_definitions(cinemaseat-audio-decoder PRIVATE CSAD_BUILD_ID="${CSAD_BUILD_ID}" CSAD_SERVICE_VERSION="${SERVICE_VERSION}")
install(TARGETS cinemaseat-audio-decoder RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_library(TEST_AVFORMAT avformat PATHS "${CSAD_DEPS}/lib" NO_DEFAULT_PATH REQUIRED)
  add_executable(fixture_export tests/integration/fixture_export.cpp)
  target_link_libraries(fixture_export PRIVATE csad_core ${TEST_AVFORMAT})
  add_executable(core_tests tests/unit/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE csad_core)
  add_test(NAME core COMMAND core_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
endif()
